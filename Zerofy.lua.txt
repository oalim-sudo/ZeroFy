local function PatchPrompt(p)
    if not (p and typeof(p) == "Instance" and p:IsA("ProximityPrompt")) then
        return
    end
    if FarmEgg.enabled then
        if FarmEgg.savedDurations[p] == nil then
            FarmEgg.savedDurations[p] = p.HoldDuration
        end
        p.HoldDuration = 0
        p.RequiresLineOfSight = false
    elseif FarmEgg.savedDurations[p] ~= nil then
        pcall(function()
            p.HoldDuration = FarmEgg.savedDurations[p]
        end)
        FarmEgg.savedDurations[p] = nil
    end
end

local function SetPromptPatch(on)
    if on then
        for _, item in ipairs(Workspace:GetDescendants()) do
            PatchPrompt(item)
        end
        table.insert(FarmEgg.patchConns, Workspace.DescendantAdded:Connect(PatchPrompt))
        table.insert(FarmEgg.patchConns, ProximityPromptService.PromptShown:Connect(PatchPrompt))
    else
        for _, c in ipairs(FarmEgg.patchConns) do
            pcall(function() c:Disconnect() end)
        end
        FarmEgg.patchConns = {}
        for p, d in pairs(FarmEgg.savedDurations) do
            if p and p.Parent then
                pcall(function() p.HoldDuration = d end)
            end
        end
        FarmEgg.savedDurations = {}
    end
end

local function GetRoot()
    local ch = LocalPlayer.Character
    if not ch then return nil, nil end
    return ch:FindFirstChildOfClass("Humanoid"), ch:FindFirstChild("HumanoidRootPart")
end

local function SaveOrigin()
    local _, root = GetRoot()
    if not root then
        FarmEgg.originCF = nil
        return false
    end
    FarmEgg.originCF = root.CFrame
    return true
end

local function ForcePosition(cf)
    local hum, root = GetRoot()
    if not root then return end
    if hum then
        hum.Sit = false
        hum.Health = math.max(hum.Health, 100)
    end
    root.CFrame = cf
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

local function DoTeleportTo(targetCF)
    local hum, root = GetRoot()
    if not hum or not root then return end
    ForcePosition(targetCF)
    task.wait(0.05)
    ForcePosition(targetCF)
end

local function DoTweenTo(targetCF)
    local hum, root = GetRoot()
    if not hum or not root then return end

    local cam = Workspace.CurrentCamera
    local oldCamCF = cam and cam.CFrame
    local oldCamType = cam and cam.CameraType

    if cam then
        cam.CameraType = Enum.CameraType.Scriptable
        cam.CFrame = oldCamCF
    end

    hum.BreakJointsOnDeath = false
    hum.PlatformStand = true
    hum.Sit = false
    hum.Health = math.max(hum.Health, 100)

    for _, d in ipairs(LocalPlayer.Character:GetDescendants()) do
        if d:IsA("Motor6D") then d.Enabled = true end
    end

    local startCF = root.CFrame
    local distance = (targetCF.Position - startCF.Position).Magnitude
    local duration = math.clamp(distance / math.max(TWEEN_SPEED, 1), 0.05, 3)

    local t0 = os.clock()
    local hb
    hb = RunService.Heartbeat:Connect(function()
        local elapsed = os.clock() - t0
        local alpha = math.clamp(elapsed / duration, 0, 1)

        if not root or not root.Parent or not hum or hum.Health <= 0 then
            if hb then hb:Disconnect() end
            return
        end

        hum.Health = math.max(hum.Health, 100)
        hum.PlatformStand = true
        root.CFrame = startCF:Lerp(targetCF, alpha)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        if cam then cam.CFrame = oldCamCF end

        if alpha >= 1 then
            if hb then hb:Disconnect() end
        end
    end)

    task.wait(duration + 0.02)

    if hb then pcall(function() hb:Disconnect() end) end

    hum.PlatformStand = false
    ForcePosition(targetCF)

    if cam then
        cam.CameraType = oldCamType or Enum.CameraType.Custom
    end
end

local function GetTargetCF()
    local success, result = pcall(function()
        local obj = Workspace:FindFirstChild("__OBJECTS")
        if obj then
            local areas = obj:FindFirstChild("Areas")
            if areas then
                if FarmEgg.helpFriend then
                    local guardAreas = areas:FindFirstChild("GuardAreas")
                    if guardAreas then
                        local forest = guardAreas:FindFirstChild("Forest")
                        if forest then
                            local reqSign = forest:FindFirstChild("RequiredSpeedSign")
                            if reqSign then
                                local main = reqSign:FindFirstChild("Main")
                                if main then
                                    if main:IsA("BasePart") then
                                        return main.CFrame
                                    elseif main:IsA("Model") then
                                        local part = main.PrimaryPart or main:FindFirstChildWhichIsA("BasePart", true)
                                        if part then return part.CFrame end
                                    end
                                end
                            end
                        end
                    end
                else
                    local startArea = areas:FindFirstChild("StartArea")
                    if startArea then
                        if startArea:IsA("BasePart") then
                            return startArea.CFrame
                        elseif startArea:IsA("Model") then
                            local part = startArea.PrimaryPart or startArea:FindFirstChildWhichIsA("BasePart", true)
                            if part then return part.CFrame end
                        end
                    end
                end
            end
        end
        return nil
    end)
    if success and result then
        return result
    end
    return CFrame.new(Vector3.new(539.7, 71.1, -364.9))
end

local function TriggerFarmEgg()
    if not FarmEgg.enabled or FarmEgg.running then return end
    FarmEgg.running = true

    if not SaveOrigin() then
        FarmEgg.running = false
        return
    end

    task.wait(0.05)

    local targetCF = GetTargetCF()
    if FarmEgg.mode == "Teleport" then
        DoTeleportTo(targetCF)
    else
        DoTweenTo(targetCF)
    end

    -- Stay at the target/start part for 0.5s before backing to the main position
    task.wait(0.5)

    if FarmEgg.helpFriend then
        pcall(function()
            if EggCmds and PlayerRequest then
                EggCmds.RequestDropHeldAreaEgg(PlayerRequest)
            end
        end)
        task.wait(0.15)
    end

    if FarmEgg.originCF then
        if FarmEgg.mode == "Teleport" then
            DoTeleportTo(FarmEgg.originCF)
        else
            DoTweenTo(FarmEgg.originCF)
        end
    end

    FarmEgg.running = false
end

local function SetEnabled(on)
    FarmEgg.enabled = on == true
    SetPromptPatch(FarmEgg.enabled)

    if FarmEgg.enabled then
        if not FarmEgg.promptConn then
            FarmEgg.promptConn = ProximityPromptService.PromptTriggered:Connect(function(prompt, plr)
                if plr ~= LocalPlayer then return end
                local n = tostring(prompt)
                if n == "CarryAreaEgg" or n:find("Egg") or n:find("Carry") then
                    TriggerFarmEgg()
                end
            end)
        end
    elseif FarmEgg.promptConn then
        pcall(function() FarmEgg.promptConn:Disconnect() end)
        FarmEgg.promptConn = nil
    end
end

            startPos = mainFrame.Position
            local connection
            connection = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if connection then connection:Disconnect() end
                end
            end)
        end
    end)

    topBar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            updateDrag(input)
        end
    end)
end
